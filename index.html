var SS_ID = "1dHpCTK4G7IlgUtHPzEcgmHONj0mwHlpZ_dfCA8yGZqo"; function doGet(e) { var ss = SpreadsheetApp.openById(SS_ID); var p = e.parameter; var dono = p.dono; // --- AÇÃO: SALVAR FUNDO --- if (p.action === "saveBackground") { var sP = ss.getSheetByName("preferencias") || ss.insertSheet("preferencias"); var dP = sP.getDataRange().getValues(); var achou = false; for (var i = 0; i < dP.length; i++) { if (dP[i][0] == dono) { sP.getRange(i+1, 2).setValue(p.fundo); achou = true; break; } } if (!achou) sP.appendRow([dono, p.fundo]); return ContentService.createTextOutput("OK"); } // --- AÇÃO: SALVAR BANNER --- if (p.action === "updateBanner") { var sB = ss.getSheetByName("banners"); var dB = sB.getDataRange().getValues(); for (var j = 0; j < dB.length; j++) { if (dB[j][0] == dono && dB[j][2] == p.nome) { sB.getRange(j+1, 4).setValue(p.imagem); break; } } return ContentService.createTextOutput("OK"); } // --- LEITURA DE DADOS --- try { var cats = ss.getSheetByName("categorias").getDataRange().getValues().filter(r => r[0] == dono).map(r => r[1]); var bans = ss.getSheetByName("banners").getDataRange().getValues().filter(r => r[0] == dono).map(r => ({ categoria: r[1], nome: r[2], imagem: r[3], link: r[4] })); var sPre = ss.getSheetByName("preferencias") || ss.insertSheet("preferencias"); var dPre = sPre.getDataRange().getValues(); var fundo = ""; for (var k = 0; k < dPre.length; k++) { if (dPre[k][0] == dono) fundo = dPre[k][1]; } var result = { categorias: cats, banners: bans, fundo: fundo }; return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({erro: err.toString()})).setMimeType(ContentService.MimeType.JSON); } }
